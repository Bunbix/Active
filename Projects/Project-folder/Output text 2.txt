        // ============================================
        // ARCHITECTURE OVERVIEW
        // ============================================
        // This calculator follows MVC-like pattern:
        // - Model: LoanCalculator class (business logic)
        // - View: DOM elements and event handlers
        // - Controller: Main module coordinating between Model and View
        //
        // Key Patterns:
        // 1. Module-level element caching for performance
        // 2. Data attributes for state management

        // 4. Precision handling for financial calculations

        // ============================================
        // MODEL: LoanCalculator Class (Business Logic)
        // ============================================
        class LoanCalculator {
            constructor() {
                this.PRECISION = 2; // Decimal precision for currency
                this.MIN_BALANCE = 0.01; // Minimum balance threshold
            }

            /**
             * Calculate periodic payment using standard amortization formula
             * P = (r * PV) / (1 - (1 + r)^-n)
             * 
             * @param {number} principal - Loan amount
             * @param {number} annualRate - Annual interest rate (decimal)
             * @param {number} paymentFrequency - Payments per year
             * @param {number} amortizationYears - Amortization period in years
             * @returns {number} Periodic payment amount
             */
            calculatePeriodicPayment(principal, annualRate, paymentFrequency, amortizationYears) {
                if (principal <= 0 || amortizationYears <= 0) return 0;
                
                const periodicRate = annualRate / paymentFrequency;
                const totalPayments = amortizationYears * paymentFrequency;
                
                // Handle zero interest rate case
                if (periodicRate === 0) {
                    return this.roundCurrency(principal / totalPayments);
                }
                
                // Standard amortization formula
                const numerator = periodicRate * Math.pow(1 + periodicRate, totalPayments);
                const denominator = Math.pow(1 + periodicRate, totalPayments) - 1;
                const payment = principal * (numerator / denominator);
                
                return this.roundCurrency(payment);
            }

            /**
             * Generate amortization schedule for fixed or variable rates
             * 
             * @param {Object} params - Calculation parameters
             * @returns {Array} Amortization schedule array
             */
            generateAmortizationSchedule(params) {
                const {
                    principal,
                    rates, // Array of {rate: decimal, periods: number}
                    paymentFrequency,
                    amortizationYears,
                    periodicPayment
                } = params;
                
                let balance = principal;
                let schedule = [];
                let period = 1;
                let totalInterest = 0;

                // Process each rate period
                rates.forEach(rateBlock => {
                    const periodicRate = rateBlock.rate / paymentFrequency;
                    
                    for (let i = 0; i < rateBlock.periods && balance > 0; i++, period++) {
                        const interest = this.roundCurrency(balance * periodicRate);
                        let principalPayment = this.roundCurrency(periodicPayment - interest);
                        
                        // Handle last payment adjustment
                        if (principalPayment > balance) {
                            principalPayment = balance;
                            periodicPayment = this.roundCurrency(principalPayment + interest);
                        }
                        
                        balance = this.roundCurrency(balance - principalPayment);
                        
                        // Clear negligible balance
                        if (Math.abs(balance) < this.MIN_BALANCE) {
                            balance = 0;
                        }
                        
                        totalInterest += interest;
                        
                        schedule.push({
                            period,
                            payment: periodicPayment,
                            principal: principalPayment,
                            interest,
                            balance,
                            annualRate: rateBlock.rate * 100
                        });
                    }
                });
                
                return {
                    schedule,
                    totalInterest: this.roundCurrency(totalInterest),
                    totalPayment: this.roundCurrency(principal + totalInterest)
                };
            }

            /**
             * Calculate rates array for variable rate scenario
             * 
             * @param {Object} params - Variable rate parameters
             * @returns {Array} Rates array for amortization
             */
            calculateVariableRates(params) {
                const {
                    initialRate,
                    newRate,
                    changeAfterYears,
                    paymentFrequency,
                    amortizationYears
                } = params;
                
                const rates = [];
                const totalPeriods = amortizationYears * paymentFrequency;
                const periodsBeforeChange = changeAfterYears * paymentFrequency;
                const periodsAfterChange = totalPeriods - periodsBeforeChange;
                
                if (periodsBeforeChange > 0) {
                    rates.push({
                        rate: initialRate,
                        periods: periodsBeforeChange
                    });
                }
                
                if (periodsAfterChange > 0) {
                    rates.push({
                        rate: newRate,
                        periods: periodsAfterChange
                    });
                }
                
                return rates;
            }

            /**
             * Round currency values to specified precision
             * 
             * @param {number} value - Value to round
             * @returns {number} Rounded value
             */
            roundCurrency(value) {
                return Math.round(value * Math.pow(10, this.PRECISION)) / Math.pow(10, this.PRECISION);
            }

            /**
             * Format period label based on frequency
             * 
             * @param {number} period - Period number
             * @param {number} frequency - Payment frequency
             * @returns {string} Formatted period label
             */
            formatPeriodLabel(period, frequency) {
                const year = Math.ceil(period / frequency);
                const paymentInYear = ((period - 1) % frequency) + 1;
                
                if (frequency === 1) {
                    return `Year ${year}`;
                } else {
                    return `Year ${year}, Payment ${paymentInYear}`;
                }
            }
        }

            function setupEventListeners() {

                setupInputSynchronization('loanAmount', 'loanAmountSlider', 'loanAmountValue', formatCurrency);
                setupInputSynchronization('fixedRate', 'fixedRateSlider', 'fixedRateValue', value => `${value}%`);
                setupInputSynchronization('loanTerm', 'loanTermSlider', 'loanTermValue', value => `${value} years`);

            function setupInputSynchronization(inputId, sliderId, displayId, formatter) {
                const input = document.getElementById(inputId);
                const slider = document.getElementById(sliderId);
                const display = document.getElementById(displayId);

                elements.amortizationPeriod.max = settings.maxAmortization;

                elements.paymentHeader.textContent = `${labels.singular} Payment`;

            // ============================================
            // PRIVATE METHODS: Core Calculations
            // ============================================

            function calculateAmortization() {
                // Show loading state
                setLoadingState(true);
                
                // Get input values
                const principal = parseFloat(elements.loanAmount.value);
                const paymentFrequency = parseInt(elements.paymentFrequency.value);
                const amortizationYears = parseInt(elements.amortizationPeriod.value);
                
                // Prepare rate parameters based on rate type
                let rates;
                if (state.currentRateType === 'fixed') {
                    const annualRate = parseFloat(elements.fixedRate.value) / 100;
                    rates = [{ rate: annualRate, periods: amortizationYears * paymentFrequency }];
                } else {
                    const initialRate = parseFloat(elements.initialRate.value) / 100;
                    const newRate = parseFloat(elements.newRate.value) / 100;
                    const changeAfterYears = parseInt(elements.rateChange.value);
                    
                    rates = state.calculator.calculateVariableRates({
                        initialRate,
                        newRate,
                        changeAfterYears,
                        paymentFrequency,
                        amortizationYears
                    });
                }
                
                // Calculate periodic payment
                const periodicPayment = state.calculator.calculatePeriodicPayment(
                    principal,
                    rates[0].rate,
                    paymentFrequency,
                    amortizationYears
                );
                
                // Generate amortization schedule
                const result = state.calculator.generateAmortizationSchedule({
                    principal,
                    rates,
                    paymentFrequency,
                    amortizationYears,
                    periodicPayment
                });
                
                // Update application state
                state.amortizationData = result.schedule;
                
                // Update UI
                updateSummary(principal, periodicPayment, result.totalInterest, amortizationYears);
                updateAmortizationTable();
                updatePaymentChart();
                updateFrequencyComparison(principal, rates[0].rate, amortizationYears);
                
                // Hide loading state
                setLoadingState(false);
            }

            function updateSummary(principal, periodicPayment, totalInterest, amortizationYears) {
                const totalPayment = principal + totalInterest;

                elements.totalInterest.textContent = formatCurrency(totalInterest);
                elements.totalPayment.textContent = formatCurrency(totalPayment);

                payoffDate.setFullYear(payoffDate.getFullYear() + amortizationYears);
                elements.payoffDate.textContent = payoffDate.toLocaleDateString('en-US', {
                    year: 'numeric',
                    month: 'short'
                });
            }

            function updateAmortizationTable() {
                const tbody = elements.amortizationTableBody;
                const frequency = parseInt(elements.paymentFrequency.value);
                
                // Clear existing rows
                tbody.innerHTML = '';
                
                // Performance optimization: Show sample of data
                const displayData = getDisplayData(state.amortizationData, 24);
                
                // Generate table rows
                displayData.forEach(item => {
                    if (item === null) {

                        separatorRow.innerHTML = `

                        tbody.appendChild(separatorRow);

            }

            function updateFrequencyComparison(principal, annualRate, amortizationYears) {
                const currentFrequency = parseInt(elements.paymentFrequency.value);
                const frequencies = [1, 2, 4, 12, 24, 26, 52];
                
                let comparisonHTML = '';
                
                frequencies.forEach(freq => {
                    const periodicPayment = state.calculator.calculatePeriodicPayment(
                        principal,
                        annualRate,
                        freq,
                        amortizationYears
                    );
                    
                    const annualPayment = periodicPayment * freq;
                    const labels = frequencyLabels[freq];
                    const isCurrent = freq === currentFrequency;
                    
                    comparisonHTML += `
                        <div class="comparison-card ${isCurrent ? 'highlight' : ''}">
                            <h4>${labels.plural}</h4>
                            <div class="comparison-value">${formatCurrency(periodicPayment)}</div>
                            <div style="color: #666; font-size: 0.9rem; margin-bottom: 8px;">
                                per payment
                            </div>
                            <div style="color: #2c3e50; font-weight: 600;">
                                ${formatCurrency(annualPayment)} annually
                            </div>
                            ${isCurrent ? 
                                '<div style="margin-top: 10px; color: #667eea; font-weight: 600; font-size: 0.9rem;">✓ Current Selection</div>' : 
                                ''
                            }
                        </div>
                    `;
                });
                
                elements.frequencyComparison.innerHTML = comparisonHTML;
            }

            function formatCurrency(value) {
                const settings = countrySettings[state.currentCountry];
                return settings.symbol + 
                       parseFloat(value).toLocaleString('en-US', {
                           minimumFractionDigits: 2,
                           maximumFractionDigits: 2
                       });
            }

            function getDisplayData(data, maxItems) {
                if (data.length <= maxItems) {
                    return data;
                }
                
                // Return first 12, separator, last 12
                const firstHalf = data.slice(0, maxItems / 2);
                const lastHalf = data.slice(-maxItems / 2);
                
                return [...firstHalf, null, ...lastHalf];
            }

            function prepareChartData(data, frequency) {
                const labels = [];
                const principal = [];
                const interest = [];
                
                // Sample data for chart (every nth item)
                const sampleRate = Math.max(1, Math.floor(data.length / 12));
                
                for (let i = 0; i < data.length; i += sampleRate) {
                    const item = data[i];
                    const year = Math.ceil((i + 1) / frequency);
                    
                    labels.push(`Year ${year}`);
                    principal.push(item.principal);
                    interest.push(item.interest);
                }
                
                return { labels, principal, interest };
            }

            function setLoadingState(isLoading) {

                    elements.calculateBtn.classList.add('loading');
                    elements.btnText.classList.add('hidden');
                    elements.btnLoading.classList.remove('hidden');
                } else {
                    elements.calculateBtn.classList.remove('loading');
                    elements.btnText.classList.remove('hidden');
                    elements.btnLoading.classList.add('hidden');
                }
            }

                let csvContent = "data:text/csv;charset=utf-8,";

                state.amortizationData.forEach(item => {
                    const frequency = parseInt(elements.paymentFrequency.value);

                    csvContent += `${periodLabel},${item.payment.toFixed(2)},${item.principal.toFixed(2)},${item.interest.toFixed(2)},${item.balance.toFixed(2)},${item.annualRate.toFixed(2)}%\n`;
                });
                
                const encodedUri = encodeURI(csvContent);
                const link = document.createElement("a");
                link.setAttribute("href", encodedUri);
                link.setAttribute("download", `amortization_schedule_${state.currentCountry}.csv`);
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);

    <script>
        // ============================================
        // ADVANCED FINANCIAL CALCULATOR IMPLEMENTATION
        // ============================================
        
        class AdvancedFinancialCalculator {
            constructor() {
                this.PRECISION = 8;
                this.DAYS_IN_YEAR = 365;
                this.MIN_BALANCE = 0.01;
            }
            
            // ============================================
            // 1. MULTIPLE COMPOUNDING METHODS
            // ============================================
            
            /**
             * Calculate effective periodic rate with different compounding frequencies
             */
            calculateEffectiveRate(annualRate, compoundingFreq, paymentFreq) {
                if (compoundingFreq === 'continuous') {
                    // Continuous compounding: e^(r) - 1
                    return Math.exp(annualRate / paymentFreq) - 1;
                }
                
                const n = parseInt(compoundingFreq);
                const p = parseInt(paymentFreq);
                
                if (n === p) {
                    // Same frequency
                    return annualRate / n;
                } else if (n > p) {
                    // More frequent compounding than payments
                    return Math.pow(1 + annualRate / n, n / p) - 1;
                } else {
                    // Less frequent compounding than payments
                    return annualRate / p;
                }
            }
            
            /**
             * Calculate Canadian mortgage payments (compound interest)
             */
            calculateCanadianMortgage(principal, annualRate, paymentFreq, amortizationYears) {
                // Canadian mortgages compound semi-annually
                const semiAnnualRate = annualRate / 2;
                const periods = amortizationYears * paymentFreq;
                
                // Calculate effective monthly rate from semi-annual compounding
                const monthlyRate = Math.pow(1 + semiAnnualRate, 1/6) - 1;
                
                if (monthlyRate === 0) {
                    return principal / periods;
                }
                
                return principal * monthlyRate / (1 - Math.pow(1 + monthlyRate, -periods));
            }
            
            // ============================================
            // 2. DATE-ACCURATE CALCULATIONS
            // ============================================
            
            /**
             * Calculate days between dates using different conventions
             */
            calculateDaysBetween(date1, date2, convention = 'actual/actual') {
                const timeDiff = Math.abs(date2 - date1);
                const daysDiff = Math.ceil(timeDiff / (1000 * 60 * 60 * 24));
                
                switch(convention) {
                    case '30/360':
                        // 30 days per month, 360 days per year
                        const d1 = date1.getDate();
                        const d2 = date2.getDate();
                        const m1 = date1.getMonth() + 1;
                        const m2 = date2.getMonth() + 1;
                        const y1 = date1.getFullYear();
                        const y2 = date2.getFullYear();
                        
                        let days = 360 * (y2 - y1) + 30 * (m2 - m1) + (d2 - d1);
                        return days;
                        
                    case 'actual/365':
                        return daysDiff;
                        
                    case 'actual/actual':
                    default:
                        return daysDiff;
                }
            }
            
            /**
             * Calculate interest for actual days
             */
            calculateDailyInterest(principal, annualRate, days, convention = 'actual/actual') {
                let daysInYear = this.DAYS_IN_YEAR;
                
                if (convention === 'actual/365') {
                    daysInYear = 365;
                } else if (convention === '30/360') {
                    daysInYear = 360;
                }
                
                return principal * annualRate * days / daysInYear;
            }
            
            // ============================================
            // 3. BALLOON PAYMENTS
            // ============================================
            
            /**
             * Calculate payments with balloon
             */
            calculateWithBalloon(principal, annualRate, paymentFreq, termYears, balloonType, balloonAmount) {
                const periods = termYears * paymentFreq;
                const periodicRate = annualRate / paymentFreq;
                
                if (balloonType === 'none') {
                    // Standard amortization
                    return this.calculateAmortizationPayment(principal, periodicRate, periods);
                } else if (balloonType === 'final') {
                    // Balloon equals remaining balance
                    const balloon = principal * Math.pow(1 + periodicRate, periods);
                    return 0; // Only balloon payment
                } else {
                    // Fixed balloon amount
                    const pvBalloon = balloonAmount / Math.pow(1 + periodicRate, periods);
                    const pvLoan = principal - pvBalloon;
                    
                    if (pvLoan <= 0) {
                        return 0;
                    }
                    
                    return this.calculateAmortizationPayment(pvLoan, periodicRate, periods);
                }
            }
            
            calculateAmortizationPayment(principal, periodicRate, periods) {
                if (periodicRate === 0) {
                    return principal / periods;
                }
                
                return principal * periodicRate / (1 - Math.pow(1 + periodicRate, -periods));
            }
            
            // ============================================
            // 4. IRREGULAR PAYMENTS
            // ============================================
            
            /**
             * Generate schedule with irregular payments
             */
            generateScheduleWithIrregular(params) {
                const {
                    principal,
                    annualRate,
                    paymentFreq,
                    amortizationYears,
                    startDate,
                    irregularPayments = [],
                    dayCountConvention = 'actual/actual'
                } = params;
                
                const schedule = [];
                let balance = principal;
                const periodicRate = annualRate / paymentFreq;
                const standardPayment = this.calculateAmortizationPayment(
                    principal, periodicRate, amortizationYears * paymentFreq
                );
                
                let currentDate = new Date(startDate);
                
                // Sort irregular payments by date
                irregularPayments.sort((a, b) => new Date(a.date) - new Date(b.date));
                
                for (let period = 1; period <= amortizationYears * paymentFreq && balance > 0; period++) {
                    // Calculate next payment date
                    let paymentDate = new Date(currentDate);
                    if (paymentFreq === 12) {
                        paymentDate.setMonth(paymentDate.getMonth() + 1);
                    } else if (paymentFreq === 24) {
                        paymentDate.setDate(paymentDate.getDate() + 15);
                    } else if (paymentFreq === 52) {
                        paymentDate.setDate(paymentDate.getDate() + 7);
                    } else if (paymentFreq === 26) {
                        paymentDate.setDate(paymentDate.getDate() + 14);
                    } else {
                        paymentDate.setMonth(paymentDate.getMonth() + 12 / paymentFreq);
                    }
                    
                    // Check for irregular payments on this date
                    let irregularAmount = 0;
                    const irregularOnDate = irregularPayments.find(p => 
                        this.isSameDate(new Date(p.date), paymentDate)
                    );
                    
                    if (irregularOnDate) {
                        irregularAmount = irregularOnDate.amount;
                    }
                    
                    // Calculate interest based on days
                    const days = this.calculateDaysBetween(currentDate, paymentDate, dayCountConvention);
                    const interest = this.calculateDailyInterest(balance, annualRate, days, dayCountConvention);
                    
                    let payment = standardPayment + irregularAmount;
                    let principalPayment = payment - interest;
                    
                    // Handle overpayment
                    if (principalPayment > balance) {
                        principalPayment = balance;
                        payment = principalPayment + interest;
                    }
                    
                    balance -= principalPayment;
                    
                    if (Math.abs(balance) < this.MIN_BALANCE) {
                        balance = 0;
                    }
                    
                    schedule.push({
                        period,
                        date: new Date(paymentDate),
                        payment: this.roundCurrency(payment),
                        principal: this.roundCurrency(principalPayment),
                        interest: this.roundCurrency(interest),
                        balance: this.roundCurrency(balance),
                        type: irregularOnDate ? 'irregular' : 'regular',
                        irregularAmount: this.roundCurrency(irregularAmount)
                    });
                    
                    currentDate = paymentDate;
                }
                
                return schedule;
            }
            
            isSameDate(date1, date2) {
                return date1.getFullYear() === date2.getFullYear() &&
                       date1.getMonth() === date2.getMonth() &&
                       date1.getDate() === date2.getDate();
            }
            
            // ============================================
            // 5. TIME VALUE OF MONEY CALCULATIONS
            // ============================================
            
            /**
             * Calculate Future Value
             */
            calculateFV(pv, rate, periods, pmt = 0, type = 0) {
                // type: 0 = end of period, 1 = beginning of period
                const r = rate;
                const n = periods;
                
                if (r === 0) {
                    return pv + pmt * n;
                }
                
                let fv = pv * Math.pow(1 + r, n);
                
                if (pmt !== 0) {
                    if (type === 0) {
                        // End of period
                        fv += pmt * (Math.pow(1 + r, n) - 1) / r;
                    } else {
                        // Beginning of period
                        fv += pmt * (Math.pow(1 + r, n) - 1) / r * (1 + r);
                    }
                }
                
                return this.roundCurrency(fv);
            }
            
            /**
             * Calculate Present Value
             */
            calculatePV(fv, rate, periods, pmt = 0, type = 0) {
                const r = rate;
                const n = periods;
                
                if (r === 0) {
                    return fv - pmt * n;
                }
                
                let pv = fv / Math.pow(1 + r, n);
                
                if (pmt !== 0) {
                    if (type === 0) {
                        // End of period
                        pv += pmt * (1 - Math.pow(1 + r, -n)) / r;
                    } else {
                        // Beginning of period
                        pv += pmt * (1 - Math.pow(1 + r, -n)) / r * (1 + r);
                    }
                }
                
                return this.roundCurrency(pv);
            }
            
            /**
             * Calculate Net Present Value
             */
            calculateNPV(cashFlows, discountRate) {
                let npv = 0;
                
                for (let i = 0; i < cashFlows.length; i++) {
                    npv += cashFlows[i] / Math.pow(1 + discountRate, i);
                }
                
                return this.roundCurrency(npv);
            }
            
            /**
             * Calculate Internal Rate of Return using Newton-Raphson method
             */
            calculateIRR(cashFlows, guess = 0.1, maxIterations = 1000, tolerance = 0.00001) {
                let x = guess;
                
                for (let i = 0; i < maxIterations; i++) {
                    const npv = this.calculateNPVForIRR(cashFlows, x);
                    const npvPrime = this.calculateNPVPrime(cashFlows, x);
                    
                    if (Math.abs(npvPrime) < tolerance) {
                        break;
                    }
                    
                    const xNew = x - npv / npvPrime;
                    
                    if (Math.abs(xNew - x) < tolerance) {
                        return xNew;
                    }
                    
                    x = xNew;
                }
                
                return x;
            }
            
            calculateNPVForIRR(cashFlows, rate) {
                let npv = 0;
                
                for (let i = 0; i < cashFlows.length; i++) {
                    npv += cashFlows[i] / Math.pow(1 + rate, i);
                }
                
                return npv;
            }
            
            calculateNPVPrime(cashFlows, rate) {
                let sum = 0;
                
                for (let i = 1; i < cashFlows.length; i++) {
                    sum += i * cashFlows[i] / Math.pow(1 + rate, i + 1);
                }
                
                return -sum;
            }
            
            // ============================================
            // 6. SCENARIO COMPARISON
            // ============================================
            
            /**
             * Compare multiple loan scenarios
             */
            compareScenarios(scenarios) {
                return scenarios.map(scenario => {
                    const periodicPayment = this.calculateAmortizationPayment(
                        scenario.amount,
                        scenario.rate / scenario.paymentFreq / 100,
                        scenario.term * scenario.paymentFreq
                    );
                    
                    const totalPayment = periodicPayment * scenario.term * scenario.paymentFreq;
                    const totalInterest = totalPayment - scenario.amount;
                    
                    return {
                        ...scenario,
                        periodicPayment: this.roundCurrency(periodicPayment),
                        totalPayment: this.roundCurrency(totalPayment),
                        totalInterest: this.roundCurrency(totalInterest),
                        interestPercentage: this.roundCurrency((totalInterest / scenario.amount) * 100)
                    };
                });
            }
            
            // ============================================
            // UTILITY METHODS
            // ============================================
            
            roundCurrency(value) {
                return Math.round(value * Math.pow(10, 2)) / Math.pow(10, 2);
            }
            
            roundPercentage(value) {
                return Math.round(value * Math.pow(10, 4)) / Math.pow(10, 2);
            }
        }

            calculateLoan() {
                // Gather inputs
                const principal = parseFloat(this.elements.loanAmount.value);
                const annualRate = parseFloat(this.elements.interestRate.value) / 100;
                const paymentFreq = parseInt(this.elements.paymentFrequency.value);
                const amortizationYears = parseInt(this.elements.amortizationYears.value || this.elements.loanTerm.value);
                const compoundingFreq = this.elements.compoundingFrequency.value;
                const startDate = new Date(this.elements.startDate.value);
                const dayCountConvention = this.elements.dayCount.value;
                const balloonType = this.elements.balloonType.value;
                const balloonAmount = parseFloat(this.elements.balloonAmount.value) || 0;
                
                // Calculate effective rate
                const effectiveRate = this.calculator.calculateEffectiveRate(
                    annualRate, 
                    compoundingFreq, 
                    paymentFreq
                );
                
                // Generate schedule
                const schedule = this.calculator.generateScheduleWithIrregular({
                    principal,
                    annualRate,
                    paymentFreq,
                    amortizationYears,
                    startDate,
                    irregularPayments: this.irregularPayments,
                    dayCountConvention
                });
                
                // Calculate totals
                const totalInterest = schedule.reduce((sum, row) => sum + row.interest, 0);
                const totalPayment = principal + totalInterest;
                const periodicPayment = schedule.length > 0 ? schedule[0].payment : 0;
                
                // Update UI
                this.elements.periodicPayment.textContent = `$${this.calculator.roundCurrency(periodicPayment).toLocaleString()}`;
                this.elements.totalInterest.textContent = `$${this.calculator.roundCurrency(totalInterest).toLocaleString()}`;
                this.elements.totalPayment.textContent = `$${this.calculator.roundCurrency(totalPayment).toLocaleString()}`;
                this.elements.effectiveRate.textContent = `${this.calculator.roundPercentage(effectiveRate * 100)}%`;
                
                // Update frequency label
                const frequency = parseInt(this.elements.paymentFrequency.value);
                const labels = {
                    1: 'Annual', 2: 'Semi-Annual', 4: 'Quarterly', 
                    12: 'Monthly', 24: 'Semi-Monthly', 26: 'Bi-Weekly', 52: 'Weekly'
                };
                this.elements.paymentFrequencyLabel.textContent = labels[frequency] || 'Periodic';
                
                // Update amortization table
                this.updateAmortizationTable(schedule);
                
                // Update chart
                this.updateLoanChart(schedule);
            }

            calculateTVM(calculationType) {
                const pv = parseFloat(this.elements.pv.value) || 0;
                const fv = parseFloat(this.elements.fv.value) || 0;
                const rate = parseFloat(this.elements.rate.value) / 100 || 0;
                const periods = parseFloat(this.elements.periods.value) || 0;
                const payment = parseFloat(this.elements.payment.value) || 0;
                const compounding = this.elements.tvmCompounding.value;
                
                // Adjust rate for compounding frequency
                let adjustedRate = rate;
                if (compounding === 'semi-annual') {
                    adjustedRate = rate / 2;
                } else if (compounding === 'quarterly') {
                    adjustedRate = rate / 4;
                } else if (compounding === 'monthly') {
                    adjustedRate = rate / 12;
                }
                
                let result;
                let description;
                
                switch(calculationType) {
                    case 'pv':
                        result = this.calculator.calculatePV(fv, adjustedRate, periods, payment, 0);
                        description = `Present Value of $${fv} at ${rate*100}% over ${periods} periods`;
                        break;
                        
                    case 'fv':
                        result = this.calculator.calculateFV(pv, adjustedRate, periods, payment, 0);
                        description = `Future Value of $${pv} at ${rate*100}% over ${periods} periods`;
                        break;
                        
                    case 'rate':
                        // Simple rate calculation for demonstration
                        if (fv > pv && periods > 0) {
                            result = (Math.pow(fv / pv, 1 / periods) - 1) * 100;
                            description = `Required rate to grow $${pv} to $${fv} in ${periods} periods`;
                        } else {
                            result = 0;
                            description = 'Invalid inputs for rate calculation';
                        }
                        break;
                        
                    case 'periods':
                        if (fv > pv && rate > 0) {
                            result = Math.log(fv / pv) / Math.log(1 + adjustedRate);
                            description = `Periods needed to grow $${pv} to $${fv} at ${rate*100}%`;
                        } else {
                            result = 0;
                            description = 'Invalid inputs for periods calculation';
                        }
                        break;
                }
                
                // Update UI
                this.elements.tvmResultValue.textContent = `$${this.calculator.roundCurrency(result).toLocaleString()}`;
                this.elements.tvmResultDescription.textContent = description;
                
                // Update chart
                this.updateTVMChart(pv, adjustedRate, periods, payment);
            }

            // CASH FLOW ANALYSIS FUNCTIONS

            addCashFlowEntry() {
                const container = this.elements.cashFlowEntries;
                const entryId = Date.now();
                
                const entryDiv = document.createElement('div');
                entryDiv.className = 'payment-item';
                entryDiv.dataset.id = entryId;
                
                entryDiv.innerHTML = `
                    <span>Period ${this.cashFlows.length + 1}</span>
                    <input type="number" class="cashflow-amount" placeholder="Amount" value="${this.cashFlows.length === 0 ? -10000 : 2000}">
                    <button class="remove-payment" data-id="${entryId}">
                        <i class="fas fa-times"></i>
                    </button>
                `;
                
                container.appendChild(entryDiv);
                
                // Add to cash flows array
                this.cashFlows.push(this.cashFlows.length === 0 ? -10000 : 2000);
                
                // Add event listeners
                entryDiv.querySelector('.cashflow-amount').addEventListener('input', (e) => {
                    const index = Array.from(container.children).indexOf(entryDiv);
                    if (index >= 0) {
                        this.cashFlows[index] = parseFloat(e.target.value) || 0;
                        this.updateCashFlowChart();
                    }
                });
                
                entryDiv.querySelector('.remove-payment').addEventListener('click', (e) => {
                    const id = e.target.closest('.remove-payment').dataset.id;
                    this.removeCashFlowEntry(id);
                });
                
                this.updateCashFlowChart();
            }
            
            removeCashFlowEntry(id) {
                const element = document.querySelector(`.payment-item[data-id="${id}"]`);
                if (element) {
                    const container = this.elements.cashFlowEntries;
                    const index = Array.from(container.children).indexOf(element);
                    
                    if (index >= 0) {
                        this.cashFlows.splice(index, 1);
                        element.remove();
                        this.updateCashFlowChart();
                    }
                }
            }
            
            calculateCashFlowNPV() {
                const discountRate = parseFloat(this.elements.discountRate.value) / 100;
                const npv = this.calculator.calculateNPV(this.cashFlows, discountRate);
                
                this.elements.npvResult.textContent = `$${npv.toLocaleString()}`;
            }
            
            calculateCashFlowIRR() {
                if (this.cashFlows.length < 2 || this.cashFlows[0] >= 0) {
                    alert('Need at least one negative (investment) cash flow followed by positive returns.');
                    return;
                }
                
                const irr = this.calculator.calculateIRR(this.cashFlows) * 100;
                this.elements.irrResult.textContent = `${this.calculator.roundPercentage(irr)}%`;
                
                // Calculate payback period
                let cumulative = 0;
                let paybackPeriod = 0;
                let remaining = -this.cashFlows[0];
                
                for (let i = 1; i < this.cashFlows.length; i++) {
                    cumulative += this.cashFlows[i];
                    if (cumulative >= remaining) {
                        paybackPeriod = i;
                        break;
                    }
                }
                
                this.elements.paybackPeriod.textContent = `${paybackPeriod} years`;
            }
            
            updateCashFlowChart() {
                const labels = this.cashFlows.map((_, i) => `Period ${i}`);
                const data = this.cashFlows;
                const colors = data.map(amount => amount >= 0 ? '#4CAF50' : '#f44336');
                
                this.charts.cashFlow.data.labels = labels;
                this.charts.cashFlow.data.datasets[0].data = data;
                this.charts.cashFlow.data.datasets[0].backgroundColor = colors;
                this.charts.cashFlow.update();
            }
